    private void Slots_Spin(object sender, EventArgs e)
    {
        int bet = (int)numBetSlots.Value;
        if (bet > balance) { ShowGameMessage(lblSlotsResult, "Недостаточно средств!", Color.Red); return; }
        balance -= bet;
        UpdateBalanceDisplay();
        SaveBalance();
        // Генерируем символы
        int[] results = new int[3];
        for (int i = 0; i < 3; i++) results[i] = rand.Next(symbols.Length);
        for (int i = 0; i < 3; i++) reels[i].Text = symbols[results[i]];
        // Выигрыш
        int win = 0;
        if (results[0] == results[1] && results[1] == results[2])
        {
            // три одинаковых
            win = bet * 10;
            ShowGameMessage(lblSlotsResult, $"ДЖЕКПОТ! Выигрыш {win}", Color.Green);
        }
        else if (results[0] == results[1] || results[1] == results[2] || results[0] == results[2])
        {
            win = bet * 2;
            ShowGameMessage(lblSlotsResult, $"Две одинаковых! Выигрыш {win}", Color.Orange);
        }
        else
        {
            ShowGameMessage(lblSlotsResult, "Ничего не выпало. Попробуйте ещё!", Color.Red);
        }
        balance += win;
        UpdateBalanceDisplay();
        SaveBalance();
    }

    // ==================== ПЛИНКО ====================
    private Button btnDrop;
    private NumericUpDown numBetPlinko;
    private Label lblPlinkoResult;
    private Panel plinkoBoard;
    private int dropX = 0; // позиция мяча по горизонтали (0..14)
    private Timer plinkoTimer;
    private int[] multipliers = { 0, 1, 2, 3, 5, 10, 0 }; // для простоты 7 ячеек внизу

    {

    private void PlinkoBoard_Paint(object sender, PaintEventArgs e)
    {
        Graphics g = e.Graphics;
        // Рисуем колышки (сетка 7x7)
        int startX = 20, startY = 20, stepX = 30, stepY = 30;
        for (int row = 0; row < 6; row++)
        {
            int offsetX = (row % 2 == 0) ? 0 : stepX / 2;
            for (int col = 0; col < 7; col++)
            {
                int x = startX + offsetX + col * stepX;
                int y = startY + row * stepY;
                e.Graphics.FillEllipse(Brushes.Brown, x - 3, y - 3, 6, 6);
            }
        }
        // Рисуем ячейки внизу
        for (int i = 0; i < 7; i++)
        {
            int x = startX + i * stepX;
            int y = startY + 6 * stepY + 10;
            e.Graphics.DrawRectangle(Pens.Black, x - 15, y, 30, 20);
            e.Graphics.DrawString(multipliers[i].ToString(), SystemFonts.Default, Brushes.Black, x - 10, y + 2);
        }
    }

    private void Plinko_Drop(object sender, EventArgs e)
    {
        int bet = (int)numBetPlinko.Value;
        if (bet > balance) { ShowGameMessage(lblPlinkoResult, "Недостаточно средств!", Color.Red); return; }
        balance -= bet;
        UpdateBalanceDisplay();
        SaveBalance();
        // Анимация падения: случайное блуждание влево-вправо
        dropX = 3; // начальная позиция по горизонтали (0..6)
        plinkoTimer = new Timer();
        plinkoTimer.Interval = 100;
        int steps = 0;
        plinkoTimer.Tick += (ts, te) =>
        {
            if (steps >= 6) // прошли 6 рядов
            {
                plinkoTimer.Stop();
                int finalSlot = dropX;
                if (finalSlot < 0) finalSlot = 0;
                if (finalSlot > 6) finalSlot = 6;
                int multiplier = multipliers[finalSlot];
                int win = bet * multiplier;
                balance += win;
                UpdateBalanceDisplay();
                SaveBalance();
                ShowGameMessage(lblPlinkoResult, $"Шарик упал в ячейку {finalSlot+1} (множитель x{multiplier}). Выигрыш: {win}", win>0?Color.Green:Color.Red);
                plinkoBoard.Invalidate();
                return;
            }
            // случайное отклонение
            int dir = rand.Next(3) - 1; // -1,0,1
            dropX += dir;
            if (dropX < 0) dropX = 0;
            if (dropX > 6) dropX = 6;
            steps++;
            // перерисовка с шариком
            plinkoBoard.Paint -= PlinkoBoard_Paint;
            plinkoBoard.Paint += PlinkoBoard_PaintWithBall;
            plinkoBoard.Invalidate();
            plinkoBoard.Paint -= PlinkoBoard_PaintWithBall;
            plinkoBoard.Paint += PlinkoBoard_Paint;
        };
        plinkoTimer.Start();
    }

    private void PlinkoBoard_PaintWithBall(object sender, PaintEventArgs e)
    {
        PlinkoBoard_Paint(sender, e);
        // Рисуем шарик в текущей позиции
        int startX = 20, stepX = 30, stepY = 30;
        int offset = (steps % 2 == 0) ? 0 : stepX / 2;
        int x = startX + offset + dropX * stepX;
        int y = 20 + steps * stepY;
        e.Graphics.FillEllipse(Brushes.Red, x - 6, y - 6, 12, 12);
    }
    private int steps = 0; // для анимации

    private void Roulette_Spin(object sender, EventArgs e)
    {
        int bet = (int)numBetRoulette.Value;
        if (bet > balance) { ShowGameMessage(lblRouletteResult, "Недостаточно средств!", Color.Red); return; }
        int resultNumber = rand.Next(0, 37); // 0-36
        lblSpinNumber.Text = resultNumber.ToString();
        // цвет и прочее
        bool isRed = (new[] { 1,3,5,7,9,12,14,16,18,19,21,23,25,27,30,32,34,36 }.Contains(resultNumber));
        bool isBlack = (resultNumber != 0 && !isRed);
        bool isEven = (resultNumber % 2 == 0 && resultNumber != 0);
        bool isOdd = (resultNumber % 2 == 1);
        bool win = false;
        string betType = cbBetType.SelectedItem.ToString();
        string detail = "";
        if (betType == "Число (1-36)")
        {
            if (int.TryParse(txtBetValue.Text, out int num) && num >= 1 && num <= 36)
                win = (resultNumber == num);
            else win = false;
            detail = $"на число {txtBetValue.Text}";
        }
        else if (betType == "Красное") { win = isRed; detail = "на красное"; }
        else if (betType == "Чёрное") { win = isBlack; detail = "на чёрное"; }
        else if (betType == "Чётное") { win = isEven; detail = "на чётное"; }
        else if (betType == "Нечётное") { win = isOdd; detail = "на нечётное"; }
        else if (betType == "1-18") { win = (resultNumber >= 1 && resultNumber <= 18); detail = "на 1-18"; }
        else if (betType == "19-36") { win = (resultNumber >= 19 && resultNumber <= 36); detail = "на 19-36"; }
        int winAmount = 0;
        if (win)
        {
            int multiplier = (betType == "Число (1-36)") ? 35 : 2;
            winAmount = bet * multiplier;
            balance += winAmount;
            ShowGameMessage(lblRouletteResult, $"Выигрыш! Ставка {detail} сыграла. +{winAmount}", Color.Green);
        }
        else
        {
            balance -= bet;
            ShowGameMessage(lblRouletteResult, $"Проигрыш. Ставка {detail} не сыграла. Выпало {resultNumber}.", Color.Red);
        }
        UpdateBalanceDisplay();
        SaveBalance();
    }

    private void UpdateBalanceDisplay()
    {
        txtBalance.Text = balance.ToString();
        if (balance <= 0) { /* можно заблокировать игры */ }
    }

    private void SaveBalance()
    {
        try { File.WriteAllText(saveFile, balance.ToString()); } catch { }
    }

    private void LoadBalance()
    {
        if (File.Exists(saveFile))
        {
            try { balance = int.Parse(File.ReadAllText(saveFile)); } catch { }
        }
        if (balance <= 0) balance = 100;
    }
