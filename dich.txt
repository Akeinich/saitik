private void btnMeeting_Click(object sender, EventArgs e)
{
    Random rand = new Random();
    int days = 365;
    int meetings = 0;
    for (int day = 0; day < days; day++)
    {
        // Время прихода каждого в минутах от 11:00 (0) до 13:00 (120)
        double t1 = rand.NextDouble() * 120;
        double t2 = rand.NextDouble() * 120;
        if (Math.Abs(t1 - t2) <= 15)
            meetings++;
    }
    double probability = (double)meetings / days;
    MessageBox.Show($"За {days} дней встретились {meetings} раз.\nВероятность: {probability:P2}");
}
