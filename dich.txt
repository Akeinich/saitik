using System;
using System.Windows.Forms;
using ComplexNumbersLib; // наша библиотека

namespace ComplexCalculator
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        // Чтение комплексного числа из текстовых полей
        private MyComplex ReadComplex(TextBox tbRe, TextBox tbIm)
        {
            double re = 0, im = 0;
            try { re = Convert.ToDouble(tbRe.Text); } catch { re = 0; }
            try { im = Convert.ToDouble(tbIm.Text); } catch { im = 0; }
            return new MyComplex(re, im);
        }

        // Вывод результата
        private void ShowResult(MyComplex z)
        {
            txtResult.Text = z.ToString();
        }

        private void btnAdd_Click(object sender, EventArgs e)
        {
            MyComplex z1 = ReadComplex(txtZ1_Re, txtZ1_Im);
            MyComplex z2 = ReadComplex(txtZ2_Re, txtZ2_Im);
            MyComplex result = MyComplex.Sum(z1, z2);
            ShowResult(result);
        }

        private void btnSub_Click(object sender, EventArgs e)
        {
            MyComplex z1 = ReadComplex(txtZ1_Re, txtZ1_Im);
            MyComplex z2 = ReadComplex(txtZ2_Re, txtZ2_Im);
            MyComplex result = MyComplex.Sub(z1, z2);
            ShowResult(result);
        }

        private void btnMul_Click(object sender, EventArgs e)
        {
            MyComplex z1 = ReadComplex(txtZ1_Re, txtZ1_Im);
            MyComplex z2 = ReadComplex(txtZ2_Re, txtZ2_Im);
            MyComplex result = z1 * z2; // используем перегруженный оператор
            ShowResult(result);
        }

        private void btnDiv_Click(object sender, EventArgs e)
        {
            try
            {
                MyComplex z1 = ReadComplex(txtZ1_Re, txtZ1_Im);
                MyComplex z2 = ReadComplex(txtZ2_Re, txtZ2_Im);
                MyComplex result = z1 / z2;
                ShowResult(result);
            }
            catch (DivideByZeroException ex)
            {
                MessageBox.Show(ex.Message);
            }
        }

        private void btnAbs_Click(object sender, EventArgs e)
        {
            MyComplex z = ReadComplex(txtZ1_Re, txtZ1_Im);
            double abs = z.Abs;
            txtResult.Text = abs.ToString("F4");
        }
    }
}
