using System;
using System.IO;

class Generator
{
    static void Main()
    {
        // Генерация первого файла (основной массив)
        using (BinaryWriter writer = new BinaryWriter(File.OpenWrite("data1.bin")))
        {
            int n = 1000000; // или меньше для теста
            writer.Write(n);
            Random rnd = new Random();
            for (int i = 0; i < n; i++)
            {
                writer.Write(rnd.Next(1, 1001));
            }
        }
        Console.WriteLine("data1.bin создан.");

        // Генерация второго файла (запросы)
        using (BinaryWriter writer = new BinaryWriter(File.OpenWrite("data2.bin")))
        {
            int m = 100000; // количество запросов
            writer.Write(m);
            Random rnd = new Random();
            for (int i = 0; i < m; i++)
            {
                writer.Write(rnd.Next(1, 1001));
            }
        }
        Console.WriteLine("data2.bin создан.");
    }
}
