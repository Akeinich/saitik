using System;
using System.Drawing;
using System.IO;
using System.Linq;
using System.Windows.Forms;

public class MainForm : Form
{
    private Label lblScore;
    private TextBox txtScore;
    private Button btnLoteika, btnBalls, btnWheel;
    private Panel gamePanel;
    private int score = 100;
    private Random rand = new Random();
    private string saveFile = "game_save.txt";

    public MainForm()
    {
        this.Text = "Тренажёр: Лотерейка | Шарики | Колесо";
        this.Size = new Size(700, 600);
        this.StartPosition = FormStartPosition.CenterScreen;
        CreateControls();
        LoadScore();
        UpdateScoreDisplay();
        ShowLoteika();
    }

    private void CreateControls()
    {
        lblScore = new Label { Text = "Счёт:", Location = new Point(20, 20), AutoSize = true };
        txtScore = new TextBox { Location = new Point(90, 18), Width = 100, ReadOnly = true, BackColor = Color.LightGray };
        this.Controls.Add(lblScore);
        this.Controls.Add(txtScore);

        btnLoteika = new Button { Text = "Лотерейка", Location = new Point(20, 60), Size = new Size(80, 30), BackColor = Color.Gold };
        btnBalls = new Button { Text = "Шарики", Location = new Point(110, 60), Size = new Size(80, 30), BackColor = Color.Gold };
        btnWheel = new Button { Text = "Колесо", Location = new Point(200, 60), Size = new Size(80, 30), BackColor = Color.Gold };
        btnLoteika.Click += (s, e) => ShowLoteika();
        btnBalls.Click += (s, e) => ShowBalls();
        btnWheel.Click += (s, e) => ShowWheel();
        this.Controls.Add(btnLoteika);
        this.Controls.Add(btnBalls);
        this.Controls.Add(btnWheel);

        gamePanel = new Panel { Location = new Point(20, 110), Size = new Size(640, 420), BorderStyle = BorderStyle.FixedSingle };
        this.Controls.Add(gamePanel);
    }

    // ------------------ Лотерейка ------------------
    private Label[] reels;
    private Button btnSpin;
    private NumericUpDown numBetLoteika;
    private Label lblLoteikaResult;
    private string[] icons = { "🍒", "🍋", "🍊", "🍉", "🔔", "⭐", "7️⃣" };

    private void ShowLoteika()
    {
        ClearGamePanel();
        Label title = new Label { Text = "ЛОТЕРЕЙКА", Location = new Point(10, 10), AutoSize = true, Font = new Font("Arial", 14, FontStyle.Bold) };
        gamePanel.Controls.Add(title);
        reels = new Label[3];
        for (int i = 0; i < 3; i++)
        {
            reels[i] = new Label
            {
                Text = "🍒",
                Font = new Font("Segoe UI", 40, FontStyle.Bold),
                Size = new Size(80, 80),
                TextAlign = ContentAlignment.MiddleCenter,
                BorderStyle = BorderStyle.Fixed3D,
                Location = new Point(50 + i * 90, 60)
            };
            gamePanel.Controls.Add(reels[i]);
        }
        Label lblBet = new Label { Text = "Очки:", Location = new Point(20, 160), AutoSize = true };
        numBetLoteika = new NumericUpDown { Minimum = 5, Maximum = 100, Value = 10, Location = new Point(90, 158), Width = 60 };
        btnSpin = new Button { Text = "Крутить", Location = new Point(170, 155), Size = new Size(100, 30), BackColor = Color.Lime };
        btnSpin.Click += Loteika_Spin;
        lblLoteikaResult = new Label { Text = "Готово", Location = new Point(20, 200), Size = new Size(300, 30) };
        gamePanel.Controls.Add(lblBet);
        gamePanel.Controls.Add(numBetLoteika);
        gamePanel.Controls.Add(btnSpin);
        gamePanel.Controls.Add(lblLoteikaResult);
    }

    private void Loteika_Spin(object sender, EventArgs e)
    {
        int bet = (int)numBetLoteika.Value;
        if (bet > score) { ShowMessage(lblLoteikaResult, "Не хватает очков!", Color.Red); return; }
        score -= bet;
        UpdateScoreDisplay();
        SaveScore();
        int[] res = new int[3];
        for (int i = 0; i < 3; i++) res[i] = rand.Next(icons.Length);
        for (int i = 0; i < 3; i++) reels[i].Text = icons[res[i]];
        int bonus = 0;
        if (res[0] == res[1] && res[1] == res[2]) { bonus = bet * 10; ShowMessage(lblLoteikaResult, $"Джекпот! Бонус {bonus}", Color.Green); }
        else if (res[0] == res[1] || res[1] == res[2] || res[0] == res[2]) { bonus = bet * 2; ShowMessage(lblLoteikaResult, $"Два одинаковых! Бонус {bonus}", Color.Orange); }
        else ShowMessage(lblLoteikaResult, "Ничего не совпало", Color.Red);
        score += bonus;
        UpdateScoreDisplay();
        SaveScore();
    }

    // ------------------ Шарики (упрощённый, без анимации) ------------------
    private NumericUpDown numBetBalls;
    private Button btnDrop;
    private Label lblBallsResult;
    private int[] multipliers = { 0, 1, 2, 3, 5, 10, 0 };

    private void ShowBalls()
    {
        ClearGamePanel();
        Label title = new Label { Text = "ШАРИКИ", Location = new Point(10, 10), AutoSize = true, Font = new Font("Arial", 14, FontStyle.Bold) };
        gamePanel.Controls.Add(title);

        // Рисуем статичное поле
        Panel field = new Panel { Location = new Point(20, 50), Size = new Size(400, 250), BackColor = Color.LightGray };
        field.Paint += (s, e) => DrawField(e.Graphics);
        gamePanel.Controls.Add(field);

        Label lblBet = new Label { Text = "Очки:", Location = new Point(20, 320), AutoSize = true };
        numBetBalls = new NumericUpDown { Minimum = 1, Maximum = 50, Value = 5, Location = new Point(80, 318), Width = 60 };
        btnDrop = new Button { Text = "Бросить", Location = new Point(160, 315), Size = new Size(100, 30), BackColor = Color.Lime };
        btnDrop.Click += (s, e) =>
        {
            int bet = (int)numBetBalls.Value;
            if (bet > score) { ShowMessage(lblBallsResult, "Не хватает очков!", Color.Red); return; }
            score -= bet;
            UpdateScoreDisplay();
            SaveScore();

            // Симуляция падения: случайные отклонения влево/вправо
            int pos = 3; // стартовая позиция (0..6)
            for (int i = 0; i < 6; i++) pos += rand.Next(3) - 1; // -1,0,+1
            pos = Math.Max(0, Math.Min(6, pos));
            int mult = multipliers[pos];
            int bonus = bet * mult;
            score += bonus;
            UpdateScoreDisplay();
            SaveScore();
            ShowMessage(lblBallsResult, $"Шарик упал в ячейку {pos+1} (множитель x{mult}). Бонус: {bonus}", bonus>0 ? Color.Green : Color.Red);
        };
        lblBallsResult = new Label { Text = "Нажмите 'Бросить'", Location = new Point(20, 360), Size = new Size(360, 30) };
        gamePanel.Controls.Add(lblBet);
        gamePanel.Controls.Add(numBetBalls);
        gamePanel.Controls.Add(btnDrop);
        gamePanel.Controls.Add(lblBallsResult);
    }

    private void DrawField(Graphics g)
    {
        int startX = 20, startY = 20, stepX = 30, stepY = 30;
        for (int row = 0; row < 6; row++)
        {
            int off = (row % 2 == 0) ? 0 : stepX / 2;
            for (int col = 0; col < 7; col++)
            {
                int x = startX + off + col * stepX;
                int y = startY + row * stepY;
                g.FillEllipse(Brushes.Brown, x - 3, y - 3, 6, 6);
            }
        }
        for (int i = 0; i < 7; i++)
        {
            int x = startX + i * stepX;
            int y = startY + 6 * stepY + 10;
            g.DrawRectangle(Pens.Black, x - 15, y, 30, 20);
            g.DrawString(multipliers[i].ToString(), SystemFonts.DefaultFont, Brushes.Black, x - 10, y + 2);
        }
    }

    // ------------------ Колесо (рулетка) ------------------
    private ComboBox cbBetType;
    private TextBox txtBetValue;
    private NumericUpDown numBetWheel;
    private Button btnSpinWheel;
    private Label lblWheelResult, lblSpinNumber;

    private void ShowWheel()
    {
        ClearGamePanel();
        Label title = new Label { Text = "КОЛЕСО", Location = new Point(10, 10), AutoSize = true, Font = new Font("Arial", 14, FontStyle.Bold) };
        gamePanel.Controls.Add(title);
        lblSpinNumber = new Label { Text = "??", Font = new Font("Arial", 48, FontStyle.Bold), Location = new Point(250, 50), Size = new Size(100, 80), TextAlign = ContentAlignment.MiddleCenter, BorderStyle = BorderStyle.FixedSingle };
        gamePanel.Controls.Add(lblSpinNumber);

        Label lblType = new Label { Text = "Тип:", Location = new Point(20, 150), AutoSize = true };
        cbBetType = new ComboBox { Location = new Point(80, 148), Width = 120, DropDownStyle = ComboBoxStyle.DropDownList };
        cbBetType.Items.AddRange(new object[] { "Число (1-36)", "Красное", "Чёрное", "Чётное", "Нечётное", "1-18", "19-36" });
        cbBetType.SelectedIndex = 0;

        Label lblVal = new Label { Text = "Значение:", Location = new Point(20, 190), AutoSize = true };
        txtBetValue = new TextBox { Location = new Point(100, 188), Width = 60, Text = "1" };

        Label lblBet = new Label { Text = "Очки:", Location = new Point(20, 230), AutoSize = true };
        numBetWheel = new NumericUpDown { Minimum = 5, Maximum = 200, Value = 10, Location = new Point(80, 228), Width = 70 };

        btnSpinWheel = new Button { Text = "Крутить", Location = new Point(180, 225), Size = new Size(100, 30), BackColor = Color.Lime };
        btnSpinWheel.Click += Wheel_Spin;

        lblWheelResult = new Label { Text = "Готово", Location = new Point(20, 280), Size = new Size(500, 40) };

        gamePanel.Controls.Add(lblType);
        gamePanel.Controls.Add(cbBetType);
        gamePanel.Controls.Add(lblVal);
        gamePanel.Controls.Add(txtBetValue);
        gamePanel.Controls.Add(lblBet);
        gamePanel.Controls.Add(numBetWheel);
        gamePanel.Controls.Add(btnSpinWheel);
        gamePanel.Controls.Add(lblWheelResult);
    }

    private void Wheel_Spin(object sender, EventArgs e)
    {
        int bet = (int)numBetWheel.Value;
        if (bet > score) { ShowMessage(lblWheelResult, "Не хватает очков!", Color.Red); return; }
        int result = rand.Next(0, 37);
        lblSpinNumber.Text = result.ToString();
        bool isRed = (new[] { 1,3,5,7,9,12,14,16,18,19,21,23,25,27,30,32,34,36 }.Contains(result));
        bool win = false;
        string betType = cbBetType.SelectedItem.ToString();
        string detail = "";
        if (betType == "Число (1-36)")
        {
            int num = int.TryParse(txtBetValue.Text, out int n) ? n : 0;
            win = (num >= 1 && num <= 36 && result == num);
            detail = $"число {num}";
        }
        else if (betType == "Красное") { win = isRed; detail = "красное"; }
        else if (betType == "Чёрное") { win = (result != 0 && !isRed); detail = "чёрное"; }
        else if (betType == "Чётное") { win = (result % 2 == 0 && result != 0); detail = "чётное"; }
        else if (betType == "Нечётное") { win = (result % 2 == 1); detail = "нечётное"; }
        else if (betType == "1-18") { win = (result >= 1 && result <= 18); detail = "1-18"; }
        else if (betType == "19-36") { win = (result >= 19 && result <= 36); detail = "19-36"; }

        int bonus = 0;
        if (win)
        {
            int mult = (betType == "Число (1-36)") ? 35 : 2;
            bonus = bet * mult;
            score += bonus;
            ShowMessage(lblWheelResult, $"Успех! Ставка на {detail} сыграла. +{bonus}", Color.Green);
        }
        else
        {
            score -= bet;
            ShowMessage(lblWheelResult, $"Неудача. Ставка на {detail} не сыграла. Выпало {result}.", Color.Red);
        }
        UpdateScoreDisplay();
        SaveScore();
    }

    // ------------------ Общие методы ------------------
    private void ClearGamePanel() { gamePanel.Controls.Clear(); }
    private void ShowMessage(Label lbl, string text, Color color) { lbl.Text = text; lbl.ForeColor = color; }
    private void UpdateScoreDisplay() { txtScore.Text = score.ToString(); }
    private void SaveScore() { try { File.WriteAllText(saveFile, score.ToString()); } catch { } }
    private void LoadScore() { if (File.Exists(saveFile)) try { score = int.Parse(File.ReadAllText(saveFile)); } catch { } if (score <= 0) score = 100; }
}
