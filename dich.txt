using System;
using Microsoft.VisualStudio.TestTools.UnitTesting;

namespace UnitTestBankAccount
{
    [TestClass]
    public class AccountTests
    {
        [TestMethod]
        public void TestCreateAccount()
        {
            string customer = "Иванов Иван";
            double balance = 150.0;
            var account = new Account(customer, balance);
            Assert.AreEqual(customer, account.Customer);
            Assert.AreEqual(balance, account.Balance);
        }

        [TestMethod]
        public void TestCredit()
        {
            var account = new Account("Петров Пётр", 100.0);
            account.Credit(50.0);
            Assert.AreEqual(150.0, account.Balance);
        }

        [TestMethod]
        [ExpectedException(typeof(ArgumentOutOfRangeException))]
        public void TestCreditWithNegativeAmount()
        {
            var account = new Account("Сидоров Сидор", 100.0);
            account.Credit(-10.0);
        }

        [TestMethod]
        public void TestDebit()
        {
            var account = new Account("Кузнецова Анна", 200.0);
            account.Debit(50.0);
            Assert.AreEqual(150.0, account.Balance);
        }

        [TestMethod]
        [ExpectedException(typeof(ArgumentOutOfRangeException))]
        public void TestDebitWithNegativeAmount()
        {
            var account = new Account("Смирнова Елена", 100.0);
            account.Debit(-5.0);
        }

        [TestMethod]
        [ExpectedException(typeof(ArgumentException))]
        public void TestDebitMoreThanBalance()
        {
            var account = new Account("Васильев Василий", 100.0);
            account.Debit(150.0);
        }
    }
}
