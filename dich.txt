using System.Collections.Generic;
using System.Linq;

// внутри класса Person:

public Person CopyFromPersonDPO(PersonDPO personDPO, List<Role> roles)
{
    this.Id = personDPO.Id;
    var role = roles.FirstOrDefault(r => r.NameRole == personDPO.Role);
    this.RoleId = role?.Id ?? 0;
    this.FirstName = personDPO.FirstName;
    this.LastName = personDPO.LastName;
    this.Birthday = personDPO.Birthday;
    return this;
}
