using System;
using System.Collections.Generic;
using System.IO;
using System.Windows.Forms;

namespace Lab21_Reminder
{
    public partial class Form1 : Form
    {
        // Список для хранения записей (дата, описание)
        private List<Tuple<DateTime, string>> events = new List<Tuple<DateTime, string>>();
        private string dataFile = "notes.bin";

        public Form1()
        {
            InitializeComponent();
            LoadData();
            ShowTodaysEvents();
            UpdateAllEventsList();
        }

        // Загрузка из бинарного файла
        private void LoadData()
        {
            if (!File.Exists(dataFile)) return;
            using (BinaryReader reader = new BinaryReader(File.OpenRead(dataFile)))
            {
                int count = reader.ReadInt32();
                events.Clear();
                for (int i = 0; i < count; i++)
                {
                    long ticks = reader.ReadInt64();
                    DateTime date = new DateTime(ticks);
                    string desc = reader.ReadString();
                    events.Add(Tuple.Create(date, desc));
                }
            }
        }

        // Сохранение в бинарный файл
        private void SaveData()
        {
            using (BinaryWriter writer = new BinaryWriter(File.OpenWrite(dataFile)))
            {
                writer.Write(events.Count);
                foreach (var e in events)
                {
                    writer.Write(e.Item1.Ticks);
                    writer.Write(e.Item2);
                }
            }
        }

        // Показать события на сегодня
        private void ShowTodaysEvents()
        {
            DateTime today = DateTime.Today;
            var todays = events.FindAll(e => e.Item1.Date == today);
            txtTodayEvents.Clear();
            foreach (var e in todays)
                txtTodayEvents.AppendText(e.Item2 + Environment.NewLine);
        }

        // Обновить список всех событий (в ListBox)
        private void UpdateAllEventsList()
        {
            listAllEvents.Items.Clear();
            foreach (var e in events)
                listAllEvents.Items.Add($"{e.Item1.ToShortDateString()}: {e.Item2}");
        }

        // Добавление новой записи с сохранением хронологического порядка
        private void btnAdd_Click(object sender, EventArgs e)
        {
            DateTime selectedDate = dtpDate.Value.Date;
            string description = txtEvent.Text.Trim();
            if (string.IsNullOrEmpty(description))
            {
                MessageBox.Show("Введите описание!");
                return;
            }

            // Вставляем в нужное место (сортировка по дате)
            var newEvent = Tuple.Create(selectedDate, description);
            int index = 0;
            while (index < events.Count && events[index].Item1 <= selectedDate)
                index++;
            events.Insert(index, newEvent);

            SaveData();            // сохраняем сразу
            ShowTodaysEvents();    // обновляем сегодняшние
            UpdateAllEventsList(); // обновляем список
            txtEvent.Clear();
            MessageBox.Show("Событие добавлено.");
        }

        // При закрытии формы сохраняем (на всякий случай, хотя уже сохраняем при добавлении)
        private void Form1_FormClosing(object sender, FormClosingEventArgs e)
        {
            SaveData();
        }
    }
}
