using System;
using System.Drawing;
using System.IO;
using System.Windows.Forms;

public class MainForm : Form
{
    // Элементы управления
    private Label lblBalance, lblBet, lblNumber, lblResult;
    private TextBox txtBalance, txtBet, txtNumber;
    private Button btnPlay, btnAddMoney;
    private Label lblMessage;

    private int balance = 100;        // начальный баланс
    private Random random = new Random();
    private string saveFile = "balance.txt";

    public MainForm()
    {
        this.Text = "Казино: Угадай число (1-10)";
        this.Size = new Size(400, 400);
        this.StartPosition = FormStartPosition.CenterScreen;
        CreateControls();
        LoadBalance();
        UpdateBalanceDisplay();
    }

    private void CreateControls()
    {
        int x = 30, y = 20, labelW = 100, fieldW = 150;

        // Баланс
        lblBalance = new Label { Text = "Баланс:", Location = new Point(x, y), AutoSize = true };
        txtBalance = new TextBox { Location = new Point(x + labelW, y), Width = fieldW, ReadOnly = true, BackColor = Color.LightGray };
        y += 35;

        // Ставка
        lblBet = new Label { Text = "Ставка:", Location = new Point(x, y), AutoSize = true };
        txtBet = new TextBox { Location = new Point(x + labelW, y), Width = fieldW, Text = "10" };
        y += 35;

        // Число (1-10)
        lblNumber = new Label { Text = "Ваше число (1-10):", Location = new Point(x, y), AutoSize = true };
        txtNumber = new TextBox { Location = new Point(x + labelW, y), Width = fieldW };
        y += 40;

        // Кнопка "Играть"
        btnPlay = new Button { Text = "Играть", Location = new Point(x, y), Size = new Size(100, 40), BackColor = Color.LightGreen };
        btnPlay.Click += BtnPlay_Click;
        this.Controls.Add(btnPlay);

        // Кнопка "Пополнить баланс"
        btnAddMoney = new Button { Text = "Пополнить (+50)", Location = new Point(x + 120, y), Size = new Size(130, 40), BackColor = Color.LightBlue };
        btnAddMoney.Click += BtnAddMoney_Click;
        this.Controls.Add(btnAddMoney);
        y += 60;

        // Результат
        lblResult = new Label { Text = "Результат:", Location = new Point(x, y), AutoSize = true };
        lblMessage = new Label
        {
            Location = new Point(x + labelW, y),
            Size = new Size(230, 60),
            BorderStyle = BorderStyle.FixedSingle,
            BackColor = Color.White,
            Text = "Готов к игре"
        };
        y += 80;

        // Добавляем все на форму
        this.Controls.Add(lblBalance);
        this.Controls.Add(txtBalance);
        this.Controls.Add(lblBet);
        this.Controls.Add(txtBet);
        this.Controls.Add(lblNumber);
        this.Controls.Add(txtNumber);
        this.Controls.Add(lblResult);
        this.Controls.Add(lblMessage);
    }

    private void BtnPlay_Click(object sender, EventArgs e)
    {
        // Проверка ставки
        if (!int.TryParse(txtBet.Text, out int bet) || bet <= 0)
        {
            ShowMessage("Ставка должна быть положительным числом!", Color.Red);
            return;
        }
        if (bet > balance)
        {
            ShowMessage("Недостаточно средств! Пополните баланс.", Color.Red);
            return;
        }

        // Проверка введённого числа
        if (!int.TryParse(txtNumber.Text, out int userNumber) || userNumber < 1 || userNumber > 10)
        {
            ShowMessage("Введите число от 1 до 10!", Color.Red);
            return;
        }

        int luckyNumber = random.Next(1, 11); // 1..10
        if (userNumber == luckyNumber)
        {
            // Выигрыш: ставка умножается на 5
            int win = bet * 5;
            balance += win;
            ShowMessage($"🎉 ПОБЕДА! Вы угадали число {luckyNumber}! Выигрыш: {win}", Color.Green);
        }
        else
        {
            // Проигрыш
            balance -= bet;
            ShowMessage($"😞 Проигрыш. Загаданное число: {luckyNumber}. Вы потеряли {bet}", Color.Red);
        }

        UpdateBalanceDisplay();
        SaveBalance();
    }

    private void BtnAddMoney_Click(object sender, EventArgs e)
    {
        balance += 50;
        UpdateBalanceDisplay();
        SaveBalance();
        ShowMessage("Баланс пополнен на 50", Color.Blue);
    }

    private void UpdateBalanceDisplay()
    {
        txtBalance.Text = balance.ToString();
        if (balance <= 0)
        {
            btnPlay.Enabled = false;
            ShowMessage("Баланс закончился! Пополните счёт.", Color.Red);
        }
        else
        {
            btnPlay.Enabled = true;
        }
    }

    private void ShowMessage(string text, Color color)
    {
        lblMessage.Text = text;
        lblMessage.ForeColor = color;
    }

    private void SaveBalance()
    {
        try
        {
            File.WriteAllText(saveFile, balance.ToString());
        }
        catch { /* не критично */ }
    }

    private void LoadBalance()
    {
        if (File.Exists(saveFile))
        {
            try
            {
                string content = File.ReadAllText(saveFile);
                if (int.TryParse(content, out int saved))
                    balance = saved;
            }
            catch { }
        }
        if (balance <= 0) balance = 100;
        UpdateBalanceDisplay();
    }
}
