using System.Collections.Generic;
using System.Linq;
using WpfApp1Demo2018.ViewModel; // для доступа к RoleViewModel

// внутри класса PersonDPO:

public PersonDPO ShallowCopy()
{
    return (PersonDPO)this.MemberwiseClone();
}

public PersonDPO CopyFromPerson(Person person, List<Role> roles)
{
    this.Id = person.Id;
    var role = roles.FirstOrDefault(r => r.Id == person.RoleId);
    this.Role = role?.NameRole ?? "Не задано";
    this.FirstName = person.FirstName;
    this.LastName = person.LastName;
    this.Birthday = person.Birthday;
    return this;
}
