Процедура ПереченьНоменклатурыКоличествоПриИзменении(Элемент)
    // Получаем табличную часть формы
    Таблица = Элементы.ПереченьНоменклатуры;
    // Получаем индекс текущей строки
    ТекИндекс = Таблица.ТекущаяСтрока;
    Если ТекИндекс <> Неопределено Тогда
        ТекущаяСтрока = Объект.ПереченьНоменклатуры[ТекИндекс];
        Если ТекущаяСтрока.Количество <> Неопределено И ТекущаяСтрока.Цена <> Неопределено Тогда
            ТекущаяСтрока.Сумма = ТекущаяСтрока.Количество * ТекущаяСтрока.Цена;
        КонецЕсли;
    КонецЕсли;
КонецПроцедуры

    СтрокаТЧ = Элементы.ПереченьНоменклатуры.ТекущиеДанные;
    Если СтрокаТЧ = Неопределено Тогда
        Возврат;
    КонецЕсли;
    
    СтрокаТЧ.Цена = РаботаСоСправочниками.РозничнаяЦена(
        СтрокаТЧ.Номенклатура, 
        Объект.Дата
    );
    
    ОбновитьИтого();

Процедура ОбновитьИтого()
    Сумма = 0;
    Для Каждого СтрокаТЧ Из Объект.ПереченьНоменклатуры Цикл
        Сумма = Сумма + СтрокаТЧ.Сумма;
    КонецЦикла;
    Итого = Сумма;

Функция РозничнаяЦена(НужнаяНоменклатура, ДатаПолучения) Экспорт
    Запрос = Новый Запрос;
    Запрос.Текст =
        "ВЫБРАТЬ ПЕРВЫЕ 1
        |    Цены.Цена КАК Цена
        |ИЗ
        |    РегистрСведений.Цены КАК Цены
        |ГДЕ
        |    Цены.Номенклатура = &Номенклатура
        |    И Цены.Период <= &ДатаПолучения
        |УПОРЯДОЧИТЬ ПО
        |    Цены.Период УБЫВ";
    
    Запрос.УстановитьПараметр("Номенклатура", НужнаяНоменклатура);
    Запрос.УстановитьПараметр("ДатаПолучения", ДатаПолучения);
    
    Результат = Запрос.Выполнить().Выбрать();
    
    Если Результат.Следующий() Тогда
        Возврат Результат.Цена;
    Иначе
        Возврат 0;
    КонецЕсли;
КонецФункции
