using System;
using System.Drawing;
using System.Windows.Forms;

// Класс комплексного числа (встроен прямо сюда)
public class MyComplex
{
    public double Re { get; set; }
    public double Im { get; set; }

    public MyComplex() : this(0, 0) { }
    public MyComplex(double re, double im) { Re = re; Im = im; }
    public MyComplex(string re, string im)
    {
        try { Re = Convert.ToDouble(re); } catch { Re = 0; }
        try { Im = Convert.ToDouble(im); } catch { Im = 0; }
    }

    public double Abs => Math.Sqrt(Re * Re + Im * Im);

    public static MyComplex Sum(MyComplex z1, MyComplex z2)
        => new MyComplex(z1.Re + z2.Re, z1.Im + z2.Im);

    public static MyComplex Sub(MyComplex z1, MyComplex z2)
        => new MyComplex(z1.Re - z2.Re, z1.Im - z2.Im);

    public static MyComplex operator *(MyComplex z1, MyComplex z2)
        => new MyComplex(z1.Re * z2.Re - z1.Im * z2.Im,
                         z1.Re * z2.Im + z1.Im * z2.Re);

    public static MyComplex operator /(MyComplex z1, MyComplex z2)
    {
        double denom = z2.Re * z2.Re + z2.Im * z2.Im;
        if (denom == 0) throw new DivideByZeroException("Деление на ноль");
        double re = (z1.Re * z2.Re + z1.Im * z2.Im) / denom;
        double im = (z1.Im * z2.Re - z1.Re * z2.Im) / denom;
        return new MyComplex(re, im);
    }

    public override string ToString()
    {
        if (Im == 0) return Re.ToString();
        if (Re == 0) return $"{Im}i";
        string sign = Im > 0 ? "+" : "";
        return $"{Re}{sign}{Im}i";
    }
}

// Главная форма
public class MainForm : Form
{
    private TextBox txtZ1_Re, txtZ1_Im, txtZ2_Re, txtZ2_Im, txtResult;
    private Button btnAdd, btnSub, btnMul, btnDiv, btnAbs;

    public MainForm()
    {
        this.Text = "Комплексный калькулятор";
        this.Size = new Size(450, 350);
        this.StartPosition = FormStartPosition.CenterScreen;
        CreateControls();
    }

    private void CreateControls()
    {
        int x = 20, y = 20, labelW = 80, fieldW = 80;

        // Первое число
        Label lblZ1 = new Label { Text = "Первое число:", Location = new Point(x, y), AutoSize = true };
        txtZ1_Re = new TextBox { Location = new Point(x + labelW, y), Width = fieldW, TextAlign = HorizontalAlignment.Right };
        Label lblPlus1 = new Label { Text = "+ i *", Location = new Point(x + labelW + fieldW + 5, y), AutoSize = true };
        txtZ1_Im = new TextBox { Location = new Point(x + labelW + fieldW + 45, y), Width = fieldW, TextAlign = HorizontalAlignment.Right };
        y += 35;

        // Второе число
        Label lblZ2 = new Label { Text = "Второе число:", Location = new Point(x, y), AutoSize = true };
        txtZ2_Re = new TextBox { Location = new Point(x + labelW, y), Width = fieldW, TextAlign = HorizontalAlignment.Right };
        Label lblPlus2 = new Label { Text = "+ i *", Location = new Point(x + labelW + fieldW + 5, y), AutoSize = true };
        txtZ2_Im = new TextBox { Location = new Point(x + labelW + fieldW + 45, y), Width = fieldW, TextAlign = HorizontalAlignment.Right };
        y += 50;

        // Кнопки операций
        btnAdd = CreateButton("Сложить", x, y, 90);
        btnSub = CreateButton("Вычесть", x + 100, y, 90);
        btnMul = CreateButton("Умножить", x + 200, y, 90);
        btnDiv = CreateButton("Разделить", x + 300, y, 90);
        y += 40;
        btnAbs = CreateButton("Модуль (первого)", x, y, 130);

        // Результат
        Label lblRes = new Label { Text = "Результат:", Location = new Point(x, y + 45), AutoSize = true };
        txtResult = new TextBox
        {
            Location = new Point(x + 80, y + 42),
            Width = 300,
            ReadOnly = true,
            BackColor = Color.LightGray,
            TextAlign = HorizontalAlignment.Right
        };

        // Добавляем всё на форму
        this.Controls.Add(lblZ1);
        this.Controls.Add(txtZ1_Re);
        this.Controls.Add(lblPlus1);
        this.Controls.Add(txtZ1_Im);
        this.Controls.Add(lblZ2);
        this.Controls.Add(txtZ2_Re);
        this.Controls.Add(lblPlus2);
        this.Controls.Add(txtZ2_Im);
        this.Controls.Add(btnAdd);
        this.Controls.Add(btnSub);
        this.Controls.Add(btnMul);
        this.Controls.Add(btnDiv);
        this.Controls.Add(btnAbs);
        this.Controls.Add(lblRes);
        this.Controls.Add(txtResult);
    }

    private Button CreateButton(string text, int x, int y, int width)
    {
        Button btn = new Button
        {
            Text = text,
            Location = new Point(x, y),
            Size = new Size(width, 35),
            BackColor = Color.LightBlue
        };
        btn.Click += Button_Click;
        this.Controls.Add(btn);
        return btn;
    }

    private MyComplex ReadComplex(TextBox reBox, TextBox imBox)
    {
        double re = 0, im = 0;
        try { re = Convert.ToDouble(reBox.Text); } catch { }
        try { im = Convert.ToDouble(imBox.Text); } catch { }
        return new MyComplex(re, im);
    }

    private void Button_Click(object sender, EventArgs e)
    {
        Button btn = (Button)sender;
        MyComplex z1 = ReadComplex(txtZ1_Re, txtZ1_Im);
        MyComplex z2 = ReadComplex(txtZ2_Re, txtZ2_Im);
        MyComplex result = null;

        try
        {
            if (btn == btnAdd) result = MyComplex.Sum(z1, z2);
            else if (btn == btnSub) result = MyComplex.Sub(z1, z2);
            else if (btn == btnMul) result = z1 * z2;
            else if (btn == btnDiv) result = z1 / z2;
            else if (btn == btnAbs) { txtResult.Text = z1.Abs.ToString("F4"); return; }

            txtResult.Text = result.ToString();
        }
        catch (DivideByZeroException ex)
        {
            MessageBox.Show(ex.Message);
        }
        catch
        {
            MessageBox.Show("Ошибка ввода");
        }
    }
}
