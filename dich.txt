Процедура Печать(МассивОбъектов, ИменаМакетов, ПараметрыПечати) Экспорт

    Если МассивОбъектов.Количество() = 0 Тогда
        Возврат;
    КонецЕсли;

    Объект = МассивОбъектов[0];
    Макет = Документы.ОказаниеУслуги.ПолучитьМакет("Печать");
    ТабДок = Новый ТабличныйДокумент;

    // Выводим шапку
    ОбластьШапка = Макет.ПолучитьОбласть("Шапка");
    ОбластьШапка.Параметры.Номер = Объект.Номер;
    ОбластьШапка.Параметры.Дата = Объект.Дата;
    ОбластьШапка.Параметры.Склад = Объект.Склад;
    ОбластьШапка.Параметры.Клиент = Объект.Клиент;
    ОбластьШапка.Параметры.Мастер = Объект.Мастер;
    ТабДок.Вывести(ОбластьШапка);

    // Считаем общую сумму
    ИтогоСумма = 0;
    Для Каждого Строка Из Объект.ПереченьНоменклатуры Цикл
        ИтогоСумма = ИтогоСумма + Строка.Сумма;
    КонецЦикла;

    // Выводим строки табличной части
    Для Каждого Строка Из Объект.ПереченьНоменклатуры Цикл
        ОбластьСтрока = Макет.ПолучитьОбласть("Строка");
        ОбластьСтрока.Параметры.НомерСтроки = Строка.НомерСтроки;
        ОбластьСтрока.Параметры.Номенклатура = Строка.Номенклатура;
        ОбластьСтрока.Параметры.Количество = Строка.Количество;
        ОбластьСтрока.Параметры.Цена = Строка.Цена;
        ОбластьСтрока.Параметры.Сумма = Строка.Сумма;
        ТабДок.Вывести(ОбластьСтрока);
    КонецЦикла;

    // Выводим итоговую строку
    ОбластьВсего = Макет.ПолучитьОбласть("Всего");
    ОбластьВсего.Параметры.ВсегоПоДокументу = ИтогоСумма;
    ТабДок.Вывести(ОбластьВсего);

    ТабДок.Показать();
