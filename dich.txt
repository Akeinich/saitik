Процедура ОбработкаПроведения(Отказ, РежимПроведения)
    
    #Область ВременнаяТаблица
    МенеджерВТ = Новый МенеджерВременныхТаблиц;
    
    ЗапросВТ = Новый Запрос;
    ЗапросВТ.МенеджерВременныхТаблиц = МенеджерВТ;
    ЗапросВТ.Текст = 
"ВЫБРАТЬ
|    ПереченьНоменклатуры.Номенклатура,
|    ПереченьНоменклатуры.Номенклатура.ВидНоменклатуры КАК ВидНоменклатуры,
|    ПереченьНоменклатуры.НаборСвойств,
|    СУММА(ПереченьНоменклатуры.Количество) КАК КоличествоВДокументе,
|    СУММА(ПереченьНоменклатуры.Сумма) КАК СуммаВДокументе,
|    0 КАК Стоимость
|ПОМЕСТИТЬ НоменклатураДокумента
|ИЗ
|    Документ.ОказаниеУслуги.ПереченьНоменклатуры КАК ПереченьНоменклатуры
|ГДЕ
|    ПереченьНоменклатуры.Ссылка = &Ссылка
|СГРУППИРОВАТЬ ПО
|    ПереченьНоменклатуры.Номенклатура,
|    ПереченьНоменклатуры.Номенклатура.ВидНоменклатуры,
|    ПереченьНоменклатуры.НаборСвойств";
    
    ЗапросВТ.УстановитьПараметр("Ссылка", Ссылка);
    ЗапросВТ.Выполнить();
    #КонецОбласти
    
    #Область Остатки
    ЗапросОстатки = Новый Запрос;
    ЗапросОстатки.МенеджерВременныхТаблиц = МенеджерВТ;
    ЗапросОстатки.Текст = 
    "ВЫБРАТЬ
    |    НоменклатураДокумента.Номенклатура,
    |    НоменклатураДокумента.ВидНоменклатуры,
    |    НоменклатураДокумента.НаборСвойств,
    |    НоменклатураДокумента.КоличествоВДокументе,
    |    НоменклатураДокумента.СуммаВДокументе,
    |    НоменклатураДокумента.Стоимость,
    |    ЕСТЬNULL(СтоимостьМатериаловОстатки.СтоимостьОстаток, 0) КАК СтоимостьОстаток,
    |    ЕСТЬNULL(ОстаткиМатериаловОстатки.КоличествоОстаток, 0) КАК КоличествоОстаток
    |ИЗ
    |    НоменклатураДокумента КАК НоменклатураДокумента
    |        ЛЕВОЕ СОЕДИНЕНИЕ РегистрНакопления.СтоимостьМатериалов.Остатки(&МоментОстатков) КАК СтоимостьМатериаловОстатки
    |            ПО НоменклатураДокумента.Номенклатура = СтоимостьМатериаловОстатки.Материал
    |        ЛЕВОЕ СОЕДИНЕНИЕ РегистрНакопления.ОстаткиМатериалов.Остатки(&МоментОстатков) КАК ОстаткиМатериаловОстатки
    |            ПО НоменклатураДокумента.Номенклатура = ОстаткиМатериаловОстатки.Материал
    |ГДЕ
    |    НоменклатураДокумента.ВидНоменклатуры = ЗНАЧЕНИЕ(Перечисление.ВидыНоменклатуры.Материал)";
    
    МоментОстатков = НачалоДня(Дата);
    ЗапросОстатки.УстановитьПараметр("МоментОстатков", МоментОстатков);
    РезультатОстатки = ЗапросОстатки.Выполнить();
    ВыборкаОстатки = РезультатОстатки.Выбрать();
    #КонецОбласти
    
    #Область Контроль
    Пока ВыборкаОстатки.Следующий() Цикл
        Если ВыборкаОстатки.КоличествоВДокументе > ВыборкаОстатки.КоличествоОстаток Тогда
            Отказ = Истина;
            Сообщить("Недостаточно материала " + ВыборкаОстатки.Номенклатура + 
                     " (вариант: " + ВыборкаОстатки.НаборСвойств + ")" +
                     ". Остаток: " + ВыборкаОстатки.КоличествоОстаток + 
                     ", требуется: " + ВыборкаОстатки.КоличествоВДокументе);
            Возврат;
        КонецЕсли;
    КонецЦикла;
    ВыборкаДвижений = РезультатОстатки.Выбрать();
    #КонецОбласти
    
    #Область Движения
    // Очищаем движения по всем регистрам, включая бухгалтерский
    Движения.ОстаткиМатериалов.Записать();
    Движения.СтоимостьМатериалов.Записать();
    Движения.Продажи.Записать();
    Движения.Управленческий.Записать();   // добавлено для бухгалтерии

    // Получаем предопределенные счета (замените "Основной" на имя вашего плана счетов, если отличается)
    СчетТовары = ПланыСчетов.Основной.НайтиПоКоду("41");
    СчетРасчетыСКлиентами = ПланыСчетов.Основной.НайтиПоКоду("62");   // Дебиторская задолженность
    СчетКапитал = ПланыСчетов.Основной.НайтиПоКоду("90");             // Капитал (доходы)
    
    Пока ВыборкаДвижений.Следующий() Цикл
        // ----- Регистры накопления (расход материалов, продажи) -----
        // Материалы
        Если ВыборкаДвижений.ВидНоменклатуры = Перечисления.ВидыНоменклатуры.Материал Тогда
            Движение = Движения.ОстаткиМатериалов.Добавить();
            Движение.ВидДвижения = ВидДвиженияНакопления.Расход;
            Движение.Период = Дата;
            Движение.Материал = ВыборкаДвижений.Номенклатура;
            Движение.НаборСвойств = ВыборкаДвижений.НаборСвойств;
            Движение.Количество = ВыборкаДвижений.КоличествоВДокументе;
            
            Движение = Движения.СтоимостьМатериалов.Добавить();
            Движение.ВидДвижения = ВидДвиженияНакопления.Расход;
            Движение.Период = Дата;
            Движение.Материал = ВыборкаДвижений.Номенклатура;
            Движение.Стоимость = ВыборкаДвижений.КоличествоВДокументе * ВыборкаДвижений.Стоимость;
        КонецЕсли;
        
        // Продажи (обороты)
        Движение = Движения.Продажи.Добавить();
        Движение.Период = Дата;
        Движение.Номенклатура = ВыборкаДвижений.Номенклатура;
        Движение.Клиенты = Клиент;
        Движение.Мастер = Мастер;
        Движение.Количество = ВыборкаДвижений.КоличествоВДокументе;
        Движение.Выручка = ВыборкаДвижений.СуммаВДокументе;
        Если ВыборкаДвижений.ВидНоменклатуры = Перечисления.ВидыНоменклатуры.Материал Тогда
            Движение.Стоимость = ВыборкаДвижений.КоличествоВДокументе * ВыборкаДвижений.Стоимость;
        Иначе
            Движение.Стоимость = 0;
        КонецЕсли;
        
        // ----- Бухгалтерские проводки (без субконто и количества) -----
        // 1. Выручка (для всех строк): Дт 62 "Дебиторская задолженность" – Кт 90 "Капитал"
        ДвижениеБух = Движения.Управленческий.Добавить();
        ДвижениеБух.Период = Дата;
        ДвижениеБух.СчетДт = СчетРасчетыСКлиентами;
        ДвижениеБух.СчетКт = СчетКапитал;
        ДвижениеБух.Сумма = ВыборкаДвижений.СуммаВДокументе;
        // ДвижениеБух.Количество = ВыборкаДвижений.КоличествоВДокументе; // отключено
        // ДвижениеБух.СубконтоДт = Клиент; // отключено
        
        // 2. Если материал – списание себестоимости: Дт 90 "Капитал" – Кт 41 "Товары"
        Если ВыборкаДвижений.ВидНоменклатуры = Перечисления.ВидыНоменклатуры.Материал Тогда
            ДвижениеБух = Движения.Управленческий.Добавить();
            ДвижениеБух.Период = Дата;
            ДвижениеБух.СчетДт = СчетКапитал;
            ДвижениеБух.СчетКт = СчетТовары;
            ДвижениеБух.Сумма = ВыборкаДвижений.КоличествоВДокументе * ВыборкаДвижений.Стоимость; // себестоимость
            // ДвижениеБух.СубконтоКт = ВыборкаДвижений.Номенклатура; // отключено
        КонецЕсли;
        
    КонецЦикла;
    
    // Запись движений всех регистров
    Движения.ОстаткиМатериалов.Записать();
    Движения.СтоимостьМатериалов.Записать();
    Движения.Продажи.Записать();
    Движения.Управленческий.Записать();   // добавлено
    #КонецОбласти
    
КонецПроцедуры
