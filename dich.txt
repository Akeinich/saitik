using System;
using System.Drawing;
using System.Windows.Forms;

public class MainForm : Form
{
    // Параметры лифта
    private int floorCount = 5;          // количество этажей (можно изменить)
    private int currentFloor = 1;        // текущий этаж
    private bool isMoving = false;       // движется ли лифт
    private int targetFloor = 1;         // целевой этаж
    private int direction = 0;           // 1 - вверх, -1 - вниз, 0 - стоит
    private Timer timer;                 // таймер для движения

    // Элементы управления
    private Panel panelElevator;         // панель, изображающая кабину
    private Label lblCurrentFloor;       // отображение текущего этажа
    private FlowLayoutPanel panelCallButtons;   // кнопки вызова на этажах
    private FlowLayoutPanel panelCabinButtons;  // кнопки в кабине
    private Button btnSetFloors;         // кнопка для изменения этажности

    // Массивы кнопок
    private Button[] floorCallButtons;
    private Button[] cabinButtons;

    public MainForm()
    {
        this.Text = "Моделирование лифта";
        this.Size = new Size(500, 600);
        this.StartPosition = FormStartPosition.CenterScreen;
        CreateControls();
        InitializeElevator(floorCount);
    }

    private void CreateControls()
    {
        // Панель для отображения лифта
        panelElevator = new Panel
        {
            Location = new Point(20, 20),
            Size = new Size(150, 150),
            BackColor = Color.Gray,
            BorderStyle = BorderStyle.FixedSingle
        };
        this.Controls.Add(panelElevator);

        // Метка текущего этажа
        lblCurrentFloor = new Label
        {
            Location = new Point(30, 180),
            Size = new Size(130, 30),
            Text = "Этаж: 1",
            Font = new Font("Arial", 12, FontStyle.Bold)
        };
        this.Controls.Add(lblCurrentFloor);

        // Кнопка для изменения этажности
        btnSetFloors = new Button
        {
            Location = new Point(20, 220),
            Size = new Size(120, 30),
            Text = "Задать этажи"
        };
        btnSetFloors.Click += BtnSetFloors_Click;
        this.Controls.Add(btnSetFloors);

        // Панель для кнопок вызова (справа)
        panelCallButtons = new FlowLayoutPanel
        {
            Location = new Point(200, 20),
            Size = new Size(120, 400),
            FlowDirection = FlowDirection.TopDown,
            AutoScroll = true,
            BorderStyle = BorderStyle.FixedSingle
        };
        this.Controls.Add(panelCallButtons);

        // Панель для кнопок в кабине (снизу)
        panelCabinButtons = new FlowLayoutPanel
        {
            Location = new Point(20, 270),
            Size = new Size(400, 200),
            FlowDirection = FlowDirection.LeftToRight,
            WrapContents = true,
            BorderStyle = BorderStyle.FixedSingle
        };
        this.Controls.Add(panelCabinButtons);

        // Таймер
        timer = new Timer();
        timer.Interval = 300; // 0.3 сек на этаж
        timer.Tick += Timer_Tick;
    }

    private void InitializeElevator(int floors)
    {
        floorCount = floors;
        currentFloor = 1;
        targetFloor = 1;
        isMoving = false;
        direction = 0;
        if (timer.Enabled) timer.Stop();

        // Создаём кнопки вызова на этажах
        floorCallButtons = new Button[floorCount + 1];
        panelCallButtons.Controls.Clear();
        for (int i = 1; i <= floorCount; i++)
        {
            Button btn = new Button
            {
                Text = i.ToString(),
                Size = new Size(80, 40),
                Tag = i,
                BackColor = Color.LightGray
            };
            btn.Click += FloorCallButton_Click;
            floorCallButtons[i] = btn;
            panelCallButtons.Controls.Add(btn);
        }

        // Создаём кнопки в кабине
        cabinButtons = new Button[floorCount + 1];
        panelCabinButtons.Controls.Clear();
        for (int i = 1; i <= floorCount; i++)
        {
            Button btn = new Button
            {
                Text = i.ToString(),
                Size = new Size(60, 60),
                Tag = i,
                Enabled = true,
                BackColor = Color.LightBlue
            };
            btn.Click += CabinButton_Click;
            cabinButtons[i] = btn;
            panelCabinButtons.Controls.Add(btn);
        }

        UpdateDisplay();
    }

    private void BtnSetFloors_Click(object sender, EventArgs e)
    {
        string input = Microsoft.VisualBasic.Interaction.InputBox(
            "Введите количество этажей (1-15):", "Этажность", floorCount.ToString());
        if (int.TryParse(input, out int newCount) && newCount >= 1 && newCount <= 15)
        {
            InitializeElevator(newCount);
        }
        else
        {
            MessageBox.Show("Некорректное число этажей (1-15)");
        }
    }

    private void FloorCallButton_Click(object sender, EventArgs e)
    {
        Button btn = (Button)sender;
        int floor = (int)btn.Tag;
        if (floor == currentFloor && !isMoving)
        {
            MessageBox.Show($"Лифт уже на этаже {floor}");
            return;
        }
        CallElevator(floor);
    }

    private void CabinButton_Click(object sender, EventArgs e)
    {
        Button btn = (Button)sender;
        int floor = (int)btn.Tag;
        if (floor == currentFloor && !isMoving)
        {
            MessageBox.Show($"Вы уже на этаже {floor}");
            return;
        }
        CallElevator(floor);
    }

    private void CallElevator(int floor)
    {
        if (isMoving)
        {
            MessageBox.Show("Лифт уже движется, подождите.");
            return;
        }
        targetFloor = floor;
        if (targetFloor > currentFloor) direction = 1;
        else if (targetFloor < currentFloor) direction = -1;
        else direction = 0;

        if (direction != 0)
        {
            isMoving = true;
            timer.Start();
            SetCabinButtonsEnabled(false);
            UpdateDisplay();
        }
    }

    private void Timer_Tick(object sender, EventArgs e)
    {
        if (!isMoving) return;

        int newFloor = currentFloor + direction;
        if (direction == 1 && newFloor <= floorCount)
        {
            currentFloor = newFloor;
        }
        else if (direction == -1 && newFloor >= 1)
        {
            currentFloor = newFloor;
        }
        else
        {
            StopElevator();
            return;
        }

        UpdateDisplay();

        if (currentFloor == targetFloor)
        {
            StopElevator();
        }
    }

    private void StopElevator()
    {
        timer.Stop();
        isMoving = false;
        direction = 0;
        SetCabinButtonsEnabled(true);
        UpdateDisplay();
        MessageBox.Show($"Лифт прибыл на этаж {currentFloor}");
    }

    private void SetCabinButtonsEnabled(bool enabled)
    {
        for (int i = 1; i <= floorCount; i++)
        {
            if (cabinButtons[i] != null)
                cabinButtons[i].Enabled = enabled;
        }
    }

    private void UpdateDisplay()
    {
        lblCurrentFloor.Text = $"Этаж: {currentFloor}";
        panelElevator.BackColor = isMoving ? Color.Yellow : Color.Gray;
        // Подсветка кнопки текущего этажа (необязательно)
        for (int i = 1; i <= floorCount; i++)
        {
            if (floorCallButtons[i] != null)
                floorCallButtons[i].BackColor = (i == currentFloor) ? Color.Green : Color.LightGray;
            if (cabinButtons[i] != null)
                cabinButtons[i].BackColor = (i == currentFloor) ? Color.Green : Color.LightBlue;
        }
    }
}
