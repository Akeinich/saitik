using System;
using System.Windows.Forms;

namespace Lab4_Branching   // замените на имя вашего проекта
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        // Задание 1: переключение текста кнопки ВКЛ/ВЫКЛ
        private void btnToggle_Click(object sender, EventArgs e)
        {
            if (btnToggle.Text == "ВКЛ")
                btnToggle.Text = "ВЫКЛ";
            else
                btnToggle.Text = "ВКЛ";
        }

        // Задание 2: перемещение метки вниз и проверка условия
        private void btnMoveDown_Click(object sender, EventArgs e)
        {
            lblMoving.Top += 10;   // смещаем на 10 пикселей вниз
            if (lblMoving.Top > 200)
            {
                lblMoving.Text = "Я опустилась ниже 200!";
            }
        }

        // Задание 3: успеет ли агент на встречу
        private void btnCheckMeeting_Click(object sender, EventArgs e)
        {
            try
            {
                double distance = Convert.ToDouble(txtDistance.Text);
                double speed = Convert.ToDouble(txtSpeed.Text);
                double timeLimit = Convert.ToDouble(txtTime.Text);

                if (speed <= 0)
                {
                    MessageBox.Show("Скорость должна быть больше нуля!");
                    return;
                }

                double timeNeeded = distance / speed;
                if (timeNeeded <= timeLimit)
                {
                    lblMeetingResult.Text = $"Агент УСПЕЕТ. В пути {timeNeeded:F2} ч, а есть {timeLimit} ч.";
                }
                else
                {
                    lblMeetingResult.Text = $"Агент НЕ УСПЕЕТ. Нужно {timeNeeded:F2} ч, а есть {timeLimit} ч.";
                }
            }
            catch
            {
                MessageBox.Show("Введите числа во все три поля (расстояние, скорость, время)");
            }
        }

        // Задание 4: день недели по номеру (switch)
        private void btnWeekday_Click(object sender, EventArgs e)
        {
            try
            {
                int day = Convert.ToInt32(txtDayNumber.Text);
                string dayName;
                switch (day)
                {
                    case 1: dayName = "Понедельник"; break;
                    case 2: dayName = "Вторник";     break;
                    case 3: dayName = "Среда";       break;
                    case 4: dayName = "Четверг";     break;
                    case 5: dayName = "Пятница";     break;
                    case 6: dayName = "Суббота";     break;
                    case 7: dayName = "Воскресенье"; break;
                    default: dayName = "Некорректный номер (1-7)"; break;
                }
                lblWeekdayResult.Text = dayName;
            }
            catch
            {
                MessageBox.Show("Введите целое число от 1 до 7");
            }
        }

        // Задание 5: определение времени года по номеру месяца (switch)
        private void btnSeason_Click(object sender, EventArgs e)
        {
            try
            {
                int month = Convert.ToInt32(txtMonth.Text);
                string season;
                switch (month)
                {
                    case 12: case 1: case 2:
                        season = "Зима";
                        break;
                    case 3: case 4: case 5:
                        season = "Весна";
                        break;
                    case 6: case 7: case 8:
                        season = "Лето";
                        break;
                    case 9: case 10: case 11:
                        season = "Осень";
                        break;
                    default:
                        season = "Некорректный номер месяца (1-12)";
                        break;
                }
                lblSeasonResult.Text = season;
            }
            catch
            {
                MessageBox.Show("Введите целое число от 1 до 12");
            }
        }
    }
}
