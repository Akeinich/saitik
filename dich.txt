using System;
using System.Windows.Forms;

namespace Lab7_Calculator
{
    public partial class Form1 : Form
    {
        private double firstOperand = 0;
        private string selectedOperator = "";
        private bool clearDisplay = false;

        public Form1()
        {
            InitializeComponent();
        }

        // Обработчик для цифр и точки
        private void DigitClick(object sender, EventArgs e)
        {
            Button btn = (Button)sender;
            if (clearDisplay)
            {
                txtDisplay.Text = "";
                clearDisplay = false;
            }
            if (btn.Text == "." && txtDisplay.Text.Contains("."))
                return; // не добавлять вторую точку
            txtDisplay.Text += btn.Text;
        }

        // Обработчик для арифметических операций (+, -, *, /)
        private void OperatorClick(object sender, EventArgs e)
        {
            Button btn = (Button)sender;
            if (!string.IsNullOrEmpty(selectedOperator))
            {
                // Если уже есть операция, вычисляем предыдущую
                Calculate();
            }
            firstOperand = Convert.ToDouble(txtDisplay.Text);
            selectedOperator = btn.Text;
            clearDisplay = true;
        }

        // Вычисление результата
        private void Calculate()
        {
            double secondOperand = Convert.ToDouble(txtDisplay.Text);
            double result = 0;
            switch (selectedOperator)
            {
                case "+": result = firstOperand + secondOperand; break;
                case "-": result = firstOperand - secondOperand; break;
                case "*": result = firstOperand * secondOperand; break;
                case "/":
                    if (secondOperand == 0)
                    {
                        MessageBox.Show("Деление на ноль!", "Ошибка");
                        txtDisplay.Text = "0";
                        return;
                    }
                    result = firstOperand / secondOperand;
                    break;
                default: return;
            }
            txtDisplay.Text = result.ToString();
            firstOperand = result;
            selectedOperator = "";
            clearDisplay = true;
        }

        // Кнопка "="
        private void btnEqual_Click(object sender, EventArgs e)
        {
            if (!string.IsNullOrEmpty(selectedOperator))
                Calculate();
            selectedOperator = "";
            clearDisplay = true;
        }

        // Кнопка "C" (очистить всё)
        private void btnClear_Click(object sender, EventArgs e)
        {
            txtDisplay.Text = "0";
            firstOperand = 0;
            selectedOperator = "";
            clearDisplay = false;
        }

        // Кнопка "√" (квадратный корень)
        private void btnSqrt_Click(object sender, EventArgs e)
        {
            double value = Convert.ToDouble(txtDisplay.Text);
            if (value < 0)
            {
                MessageBox.Show("Нельзя извлечь корень из отрицательного числа");
                return;
            }
            double result = Math.Sqrt(value);
            txtDisplay.Text = result.ToString();
            clearDisplay = true;
        }

        // Кнопка "+/-" (смена знака)
        private void btnSign_Click(object sender, EventArgs e)
        {
            double value = Convert.ToDouble(txtDisplay.Text);
            value = -value;
            txtDisplay.Text = value.ToString();
        }

        // Кнопка "1/x"
        private void btnReciprocal_Click(object sender, EventArgs e)
        {
            double value = Convert.ToDouble(txtDisplay.Text);
            if (value == 0)
            {
                MessageBox.Show("Деление на ноль невозможно");
                return;
            }
            double result = 1 / value;
            txtDisplay.Text = result.ToString();
            clearDisplay = true;
        }
    }
}
