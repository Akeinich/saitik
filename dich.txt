using System;
using System.IO;

namespace Lab13_DataProcessing
{
    class Program
    {
        static void Main(string[] args)
        {
            string file1 = "data1.bin";   // первый файл (основной)
            string file2 = "data2.bin";   // второй файл (запросы)
            string output = "result.txt"; // выходной текстовый файл

            if (!File.Exists(file1) || !File.Exists(file2))
            {
                Console.WriteLine("Один из входных файлов не найден.");
                Console.ReadKey();
                return;
            }

            try
            {
                // Массив для подсчёта частот чисел от 1 до 1000
                int[] count = new int[1001];

                // Читаем первый файл
                using (BinaryReader reader = new BinaryReader(File.OpenRead(file1)))
                {
                    int n = reader.ReadInt32(); // количество чисел
                    for (int i = 0; i < n; i++)
                    {
                        int val = reader.ReadInt32();
                        if (val >= 1 && val <= 1000)
                            count[val]++;
                        else
                            Console.WriteLine($"Число {val} вне диапазона 1..1000, пропущено.");
                    }
                }

                // Вычисляем префиксные суммы и общее количество
                long[] prefix = new long[1001];
                long total = 0;
                for (int i = 1; i <= 1000; i++)
                {
                    total += count[i];
                    prefix[i] = prefix[i - 1] + count[i];
                }

                // Обрабатываем второй файл и пишем результат
                using (BinaryReader reader = new BinaryReader(File.OpenRead(file2)))
                using (StreamWriter writer = new StreamWriter(output))
                {
                    int m = reader.ReadInt32(); // количество запросов
                    for (int i = 0; i < m; i++)
                    {
                        int x = reader.ReadInt32();
                        if (x < 1 || x > 1000)
                        {
                            writer.WriteLine("0 0 0");
                            continue;
                        }
                        long equal = count[x];
                        long less = prefix[x - 1];
                        long greater = total - prefix[x];
                        writer.WriteLine($"{equal} {less} {greater}");
                    }
                }

                Console.WriteLine($"Обработка завершена. Результат в файле {output}");
            }
            catch (Exception ex)
            {
                Console.WriteLine("Ошибка: " + ex.Message);
            }

            Console.WriteLine("Нажмите любую клавишу для выхода...");
            Console.ReadKey();
        }
    }
}
