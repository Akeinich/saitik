Процедура ОбработкаПроведения(Отказ, РежимПроведения)
    
    // Очищаем старые движения (на случай перепроведения)
    Движения.ОстаткиМатериалов.Записать();
    Движения.СтоимостьМатериалов.Записать();
    
    // Обходим все строки табличной части "ПереченьНоменклатуры"
    Для Каждого ТекущаяСтрока Из ПереченьНоменклатуры Цикл
        
        // Нас интересуют только материалы (услуги не списываем)
        Если ТекущаяСтрока.Номенклатура.ВидНоменклатуры <> Перечисления.ВидыНоменклатуры.Материал Тогда
            Продолжить;   // переходим к следующей строке
        КонецЕсли;
        
        // ---------- Регистр "ОстаткиМатериалов" (списание количества) ----------
        Движение = Движения.ОстаткиМатериалов.Добавить();
        Движение.ВидДвижения = ВидДвиженияНакопления.Расход; // списание
        Движение.Период = Дата;
        Движение.Материал = ТекущаяСтрока.Номенклатура;
        Движение.Количество = ТекущаяСтрока.Количество;
        
        // ---------- Регистр "СтоимостьМатериалов" (списание стоимости) ----------
        Движение = Движения.СтоимостьМатериалов.Добавить();
        Движение.ВидДвижения = ВидДвиженияНакопления.Расход;
        Движение.Период = Дата;
        Движение.Материал = ТекущаяСтрока.Номенклатура;
        // Стоимость = Количество * Стоимость (цена за единицу)
        Движение.Стоимость = ТекущаяСтрока.Количество * ТекущаяСтрока.Стоимость;
        
    КонецЦикла;
    
    // Сохраняем движения в регистры
    Движения.ОстаткиМатериалов.Записать();
    Движения.СтоимостьМатериалов.Записать();
