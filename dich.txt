using System;
using System.Drawing;
using System.Windows.Forms;

namespace Lab10_Shot
{
    public partial class Form1 : Form
    {
        private double v0, angleRad, vx, vy, x, y, t;
        private const double dt = 0.05;   // шаг времени (сек)
        private const double g = 9.8;      // ускорение (можно 5 для замедления)
        private bool isFlying = false;

        // Начальная позиция снаряда (в пикселях)
        private int startX = 50;
        private int startY = 380;   // чуть выше земли

        public Form1()
        {
            InitializeComponent();
            // Установка начальной позиции снаряда
            pbProjectile.Location = new Point(startX, startY);
        }

        private void btnFire_Click(object sender, EventArgs e)
        {
            if (isFlying) return; // уже летит

            try
            {
                v0 = Convert.ToDouble(txtSpeed.Text);
                double angleDeg = Convert.ToDouble(txtAngle.Text);
                angleRad = angleDeg * Math.PI / 180.0;

                if (v0 <= 0 || angleDeg <= 0 || angleDeg >= 90)
                {
                    MessageBox.Show("Угол должен быть от 0 до 90, скорость > 0");
                    return;
                }

                // Начальные скорости
                vx = v0 * Math.Cos(angleRad);
                vy = v0 * Math.Sin(angleRad);

                // Начальные координаты (в метрах), но мы будем пересчитывать в пиксели
                // Для простоты: пусть 1 пиксель = 1 метр (но можно масштабировать)
                x = 0;   // смещение от стартовой точки в метрах
                y = 0;
                t = 0;

                // Устанавливаем снаряд в стартовую позицию
                pbProjectile.Location = new Point(startX, startY);
                isFlying = true;
                timer1.Enabled = true;
                btnFire.Enabled = false;
            }
            catch
            {
                MessageBox.Show("Введите корректные числа");
            }
        }

        private void timer1_Tick(object sender, EventArgs e)
        {
            if (!isFlying) return;

            // Шаг по времени
            t += dt;

            // Новые координаты в метрах
            x = vx * t;
            y = vy * t - 0.5 * g * t * t;

            // Переводим в экранные пиксели (сдвиг от стартовой точки)
            int screenX = startX + (int)x;
            int screenY = startY - (int)y;   // ось Y в Windows идёт вниз, поэтому минус

            // Проверка попадания в землю или за пределы формы
            if (screenY >= pbGround.Top - pbProjectile.Height || screenX >= pbTarget.Left + pbTarget.Width)
            {
                // Снаряд упал или улетел
                isFlying = false;
                timer1.Enabled = false;
                btnFire.Enabled = true;

                // Проверка попадания в мишень
                Rectangle projectileRect = new Rectangle(screenX, screenY, pbProjectile.Width, pbProjectile.Height);
                Rectangle targetRect = new Rectangle(pbTarget.Left, pbTarget.Top, pbTarget.Width, pbTarget.Height);
                if (projectileRect.IntersectsWith(targetRect))
                    MessageBox.Show("Попадание в цель!");
                else
                    MessageBox.Show("Мимо! Снаряд упал.");
                return;
            }

            // Перемещаем снаряд
            pbProjectile.Location = new Point(screenX, screenY);
        }
    }
}
