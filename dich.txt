using System.Collections.Generic;
using System.Collections.ObjectModel;
using System.Windows;
using WpfApp1Demo2018.Helper;
using WpfApp1Demo2018.Model;
using WpfApp1Demo2018.ViewModel;

namespace WpfApp1Demo2018.View
{
    public partial class WindowEmployee : Window
    {
        public WindowEmployee()
        {
            InitializeComponent();
            PersonViewModel vmPerson = new PersonViewModel();
            RoleViewModel vmRole = new RoleViewModel();

            // Преобразуем ObservableCollection<Role> в List<Role> для использования Find
            List<Role> roles = new List<Role>();
            foreach (Role r in vmRole.ListRole)
                roles.Add(r);

            ObservableCollection<PersonDPO> persons = new ObservableCollection<PersonDPO>();
            FindRole finder;
            foreach (var p in vmPerson.ListPerson)
            {
                finder = new FindRole(p.RoleId);
                Role rol = roles.Find(new System.Predicate<Role>(finder.RolePredicate));
                persons.Add(new PersonDPO
                {
                    Id = p.Id,
                    Role = rol?.NameRole ?? "Не задано",
                    FirstName = p.FirstName,
                    LastName = p.LastName,
                    Birthday = p.Birthday
                });
            }
            lvEmployee.ItemsSource = persons;
        }
    }
}
