using System;
using System.IO;

namespace Lab11_CountingSort
{
    class Program
    {
        static void Main(string[] args)
        {
            string inputFile = "input.txt";
            string outputFile = "output.txt";

            // Проверяем, существует ли входной файл
            if (!File.Exists(inputFile))
            {
                Console.WriteLine("Ошибка: файл input.txt не найден!");
                Console.WriteLine("Поместите его в папку с программой.");
                Console.ReadKey();
                return;
            }

            try
            {
                // Читаем все строки из файла
                string[] lines = File.ReadAllLines(inputFile);
                if (lines.Length == 0)
                {
                    Console.WriteLine("Файл пуст.");
                    return;
                }

                // Первая строка – количество чисел
                int n = int.Parse(lines[0]);
                if (n < 1 || n > 1000000)
                {
                    Console.WriteLine("Количество чисел должно быть от 1 до 1 000 000.");
                    return;
                }

                // Массив для подсчёта (индексы 1..1000)
                int[] count = new int[1001]; // 0-й индекс не используем

                // Обрабатываем следующие n строк
                for (int i = 1; i <= n && i < lines.Length; i++)
                {
                    int value = int.Parse(lines[i]);
                    if (value < 1 || value > 1000)
                    {
                        Console.WriteLine($"Число {value} вне диапазона 1..1000, пропускаем.");
                        continue;
                    }
                    count[value]++;
                }

                // Записываем результат
                using (StreamWriter writer = new StreamWriter(outputFile))
                {
                    for (int num = 1; num <= 1000; num++)
                    {
                        for (int c = 0; c < count[num]; c++)
                        {
                            writer.WriteLine(num);
                        }
                    }
                }

                Console.WriteLine($"Сортировка завершена. Результат в файле {outputFile}");
            }
            catch (Exception ex)
            {
                Console.WriteLine("Ошибка: " + ex.Message);
            }

            Console.WriteLine("Нажмите любую клавишу для выхода...");
            Console.ReadKey();
        }
    }
}
