using System;
using System.Windows.Forms;

namespace Lab3_Resistors_Text   // замените на имя вашего проекта, если другое
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        // Вспомогательная функция для чтения значений из текстовых полей
        private bool ReadValues(out double r1, out double r2, out double r3)
        {
            r1 = r2 = r3 = 0;
            try
            {
                r1 = Convert.ToDouble(txtR1.Text);
                r2 = Convert.ToDouble(txtR2.Text);
                r3 = Convert.ToDouble(txtR3.Text);
                if (r1 < 0 || r2 < 0 || r3 < 0)
                {
                    MessageBox.Show("Сопротивления не могут быть отрицательными!");
                    return false;
                }
                return true;
            }
            catch
            {
                MessageBox.Show("Введите корректные числовые значения сопротивлений (используйте запятую для дробной части)!");
                return false;
            }
        }

        // Последовательное соединение
        private void btnSeq_Click(object sender, EventArgs e)
        {
            if (!ReadValues(out double r1, out double r2, out double r3)) return;
            double total = r1 + r2 + r3;
            txtResult.Text = total.ToString("0.##") + " Ом";
            lblSchemeDesc.Text = "Последовательное соединение: Rобщ = R1 + R2 + R3 = " + 
                                 r1 + " + " + r2 + " + " + r3 + " = " + total.ToString("0.##") + " Ом";
        }

        // Параллельное соединение
        private void btnPar_Click(object sender, EventArgs e)
        {
            if (!ReadValues(out double r1, out double r2, out double r3)) return;
            if (r1 == 0 || r2 == 0 || r3 == 0)
            {
                MessageBox.Show("Для параллельного соединения сопротивления не могут быть нулевыми!");
                return;
            }
            double total = 1 / (1 / r1 + 1 / r2 + 1 / r3);
            txtResult.Text = total.ToString("0.##") + " Ом";
            lblSchemeDesc.Text = "Параллельное соединение: 1/Rобщ = 1/R1 + 1/R2 + 1/R3\n" +
                                 "Rобщ = " + total.ToString("0.##") + " Ом";
        }

        // Смешанное соединение (R1 последовательно с (R2 || R3))
        private void btnMix_Click(object sender, EventArgs e)
        {
            if (!ReadValues(out double r1, out double r2, out double r3)) return;
            if (r2 == 0 || r3 == 0)
            {
                MessageBox.Show("R2 и R3 не могут быть нулевыми для параллельного участка!");
                return;
            }
            double parallel = 1 / (1 / r2 + 1 / r3);
            double total = r1 + parallel;
            txtResult.Text = total.ToString("0.##") + " Ом";
            lblSchemeDesc.Text = "Смешанное соединение: R1 + (R2 || R3)\n" +
                                 "R2||R3 = " + parallel.ToString("0.##") + " Ом\n" +
                                 "Rобщ = " + r1 + " + " + parallel.ToString("0.##") + " = " + total.ToString("0.##") + " Ом";
        }
    }
}
