using System;

namespace UnitTestBankAccount
{
    public class Account
    {
        public string Customer { get; private set; }
        public double Balance { get; private set; }

        public Account(string customer, double balance)
        {
            Customer = customer;
            Balance = balance;
        }

        public void Credit(double amount)
        {
            if (amount < 0)
                throw new ArgumentOutOfRangeException(nameof(amount), "Сумма не может быть отрицательной");
            Balance += amount;
        }

        public void Debit(double amount)
        {
            if (amount < 0)
                throw new ArgumentOutOfRangeException(nameof(amount), "Сумма не может быть отрицательной");
            if (amount > Balance)
                throw new ArgumentException("Недостаточно средств");
            Balance -= amount;
        }
    }
}
