        // Вычисление расстояния (s = v * t)
        private void btnCalcS_Click(object sender, EventArgs e)
        {
            try
            {
                double v = Convert.ToDouble(txtV.Text);
                double t = Convert.ToDouble(txtT.Text);
                double s = v * t;
                txtS.Text = Convert.ToString(s);
            }
            catch
            {
                MessageBox.Show("Введите корректные числа в поля скорости и времени!");
            }
        }

        // Вычисление времени (t = s / v)
        private void btnCalcT_Click(object sender, EventArgs e)
        {
            try
            {
                double s = Convert.ToDouble(txtS.Text);
                double v = Convert.ToDouble(txtV.Text);
                if (v == 0)
                {
                    MessageBox.Show("Скорость не может быть нулевой для вычисления времени!");
                    return;
                }
                double t = s / v;
                txtT.Text = Convert.ToString(t);
            }
            catch
            {
                MessageBox.Show("Введите корректные числа в поля расстояния и скорости!");
            }
        }

        // Вычисление скорости (v = s / t)
        private void btnCalcV_Click(object sender, EventArgs e)
        {
            try
            {
                double s = Convert.ToDouble(txtS.Text);
                double t = Convert.ToDouble(txtT.Text);
                if (t == 0)
                {
                    MessageBox.Show("Время не может быть нулевым для вычисления скорости!");
                    return;
                }
                double v = s / t;
                txtV.Text = Convert.ToString(v);
            }
            catch
            {
                MessageBox.Show("Введите корректные числа в поля расстояния и времени!");
            }
        }
