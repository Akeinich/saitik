using System;
using System.Windows.Forms;

namespace Lab9_ElevatorDoor
{
    public partial class Form1 : Form
    {
        private enum DoorState { Opening, Open, Closing, Closed }
        private DoorState currentState = DoorState.Closed;
        private int delayCounter = 0;   // счётчик для паузы (1 секунда = 100 тиков таймера, т.к. Interval=10)

        public Form1()
        {
            InitializeComponent();
            // Начальное состояние: створки полностью закрыты
            doorLeft.Width = 100;
            doorRight.Width = 100;
            doorRight.Left = 100;      // правая створка прижата к правому краю
            currentState = DoorState.Closed;
            btnClose.Enabled = false;   // неактивна, пока дверь не открыта
        }

        private void timer1_Tick(object sender, EventArgs e)
        {
            switch (currentState)
            {
                case DoorState.Opening:
                    // Открываем левую створку (уменьшаем ширину)
                    if (doorLeft.Width > 8)   // минимальная ширина 8 пикселей
                    {
                        doorLeft.Width -= 2;
                        // при открытии левой створки правая не двигается, но чтобы щель была посередине,
                        // можно сдвигать правую вправо, но для простоты оставим симметричное открытие:
                        // на самом деле для реалистичности можно сдвигать правую створку вправо.
                        // Сделаем простое решение: левая створка уменьшается, правая остаётся на месте.
                        // Для красоты можно также увеличивать отступ правой.
                        doorRight.Left += 1;
                        doorRight.Width -= 1;  // и правая тоже уменьшается
                    }
                    else
                    {
                        // Открытие завершено
                        currentState = DoorState.Open;
                        delayCounter = 0;
                        btnClose.Enabled = true;
                        btnOpen.Enabled = false;
                    }
                    break;

                case DoorState.Open:
                    // Ждём 1 секунду (1000 мс / 10 мс = 100 тиков)
                    delayCounter++;
                    if (delayCounter >= 100)  // 100 * 10 мс = 1 секунда
                    {
                        currentState = DoorState.Closing;
                        delayCounter = 0;
                    }
                    break;

                case DoorState.Closing:
                    // Закрываем створки
                    if (doorLeft.Width < 100)
                    {
                        doorLeft.Width += 2;
                        doorRight.Left -= 1;
                        doorRight.Width += 1;
                    }
                    else
                    {
                        // Закрытие завершено
                        currentState = DoorState.Closed;
                        timer1.Enabled = false;
                        btnOpen.Enabled = true;
                        btnClose.Enabled = false;
                        doorLeft.Width = 100;
                        doorRight.Width = 100;
                        doorRight.Left = 100;
                    }
                    break;
            }
        }

        private void btnOpen_Click(object sender, EventArgs e)
        {
            if (currentState == DoorState.Closed)
            {
                currentState = DoorState.Opening;
                timer1.Enabled = true;
                btnOpen.Enabled = false;
            }
        }

        private void btnClose_Click(object sender, EventArgs e)
        {
            if (currentState == DoorState.Open)
            {
                currentState = DoorState.Closing;
                timer1.Enabled = true;
                btnClose.Enabled = false;
                btnOpen.Enabled = false;
            }
        }
    }
}
