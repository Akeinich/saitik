using System;
using System.Drawing;
using System.Windows.Forms;

namespace Lab20_Graphics
{
    public partial class Form1 : Form
    {
        private Bitmap canvasBitmap;   // для сохранения рисунка
        private bool isDrawing = false;
        private Point lastPoint;

        public Form1()
        {
            InitializeComponent();
            // Создаём пустое изображение размером с панель
            canvasBitmap = new Bitmap(drawingPanel.Width, drawingPanel.Height);
            drawingPanel.Image = canvasBitmap;
            DrawExampleShapes();       // рисуем примеры фигур
        }

        // Метод для рисования примера фигур
        private void DrawExampleShapes()
        {
            using (Graphics g = Graphics.FromImage(canvasBitmap))
            {
                g.Clear(Color.White);  // очищаем

                // Линия красного цвета толщиной 2
                using (Pen redPen = new Pen(Color.Red, 2))
                {
                    g.DrawLine(redPen, 50, 50, 200, 50);
                }

                // Прямоугольник синим цветом (только контур)
                using (Pen bluePen = new Pen(Color.Blue, 2))
                {
                    g.DrawRectangle(bluePen, 50, 80, 150, 100);
                }

                // Закрашенный эллипс зелёным
                using (SolidBrush greenBrush = new SolidBrush(Color.Green))
                {
                    g.FillEllipse(greenBrush, 250, 50, 150, 100);
                }

                // Текст
                using (Font font = new Font("Arial", 12))
                using (SolidBrush blackBrush = new SolidBrush(Color.Black))
                {
                    g.DrawString("Рисование мышью", font, blackBrush, 250, 200);
                }
            }
            drawingPanel.Invalidate(); // обновляем отображение
        }

        // Очистка панели
        private void btnClear_Click(object sender, EventArgs e)
        {
            using (Graphics g = Graphics.FromImage(canvasBitmap))
            {
                g.Clear(Color.White);
            }
            drawingPanel.Invalidate();
        }

        // Начало рисования
        private void drawingPanel_MouseDown(object sender, MouseEventArgs e)
        {
            if (e.Button == MouseButtons.Left)
            {
                isDrawing = true;
                lastPoint = e.Location;
            }
        }

        // Рисование при движении мыши
        private void drawingPanel_MouseMove(object sender, MouseEventArgs e)
        {
            if (isDrawing)
            {
                using (Graphics g = Graphics.FromImage(canvasBitmap))
                {
                    using (Pen pen = new Pen(Color.Black, 2))
                    {
                        g.DrawLine(pen, lastPoint, e.Location);
                    }
                }
                lastPoint = e.Location;
                drawingPanel.Invalidate(); // обновляем
            }
        }

        // Окончание рисования
        private void drawingPanel_MouseUp(object sender, MouseEventArgs e)
        {
            if (e.Button == MouseButtons.Left)
                isDrawing = false;
        }

        // Если панель изменяет размер, нужно пересоздать Bitmap (необязательно для этой работы)
        private void drawingPanel_Resize(object sender, EventArgs e)
        {
            // Для простоты не обрабатываем, так как размер фиксирован
        }
    }
}
