# ascii_player.py
import os
import sys
import time
import glob

# пробуем импортировать pygame для звука
try:
    import pygame
    SOUND_AVAILABLE = True
except ImportError:
    SOUND_AVAILABLE = False
    print("Предупреждение: pygame не установлен — звук не будет воспроизводиться.")

# настройки
FRAME_INTERVAL = 1 / 30.75   # ~30.75 FPS
FRAMES_DIR = "ascii_frames"   # папка с текстовыми кадрами

def load_frames():
    """Загружает все кадры в память (может быть много, но для школьного ПК ок)."""
    frame_files = sorted(glob.glob(os.path.join(FRAMES_DIR, "frame_*.txt")))
    if not frame_files:
        print(f"Ошибка: не найдены файлы кадров в папке {FRAMES_DIR}")
        sys.exit(1)
    
    frames = []
    for fpath in frame_files:
        with open(fpath, "r", encoding="utf-8") as f:
            frames.append(f.read())
    return frames

def play_video(frames):
    """Выводит кадры один за другим в stdout (для окна Run IDE)."""
    timer = time.perf_counter
    next_frame_time = timer()
    
    for frame in frames:
        # очищаем экран (не через system, а просто перезаписываем)
        # в IDLE/Run это не сработает красиво, но в консоли будет нормально
        sys.stdout.write("\r" + frame)
        sys.stdout.flush()
        
        next_frame_time += FRAME_INTERVAL
        sleep_time = next_frame_time - timer()
        if sleep_time > 0:
            time.sleep(sleep_time)

def play_audio():
    if not SOUND_AVAILABLE:
        return
    pygame.init()
    pygame.mixer.init()
    try:
        pygame.mixer.music.load("audio.mp3")
        pygame.mixer.music.play()
    except Exception as e:
        print(f"Не удалось загрузить/воспроизвести аудио: {e}")

def main():
    # проверяем наличие папки с кадрами
    if not os.path.exists(FRAMES_DIR):
        print(f"Папка '{FRAMES_DIR}' не найдена. Убедитесь, что вы скопировали её с флешки.")
        return
    
    print("Загрузка кадров...")
    frames = load_frames()
    print(f"Загружено {len(frames)} кадров. Начинаем воспроизведение...")
    
    # запускаем звук (отдельным потоком, чтобы не блокировать видео)
    if SOUND_AVAILABLE:
        import threading
        audio_thread = threading.Thread(target=play_audio, daemon=True)
        audio_thread.start()
    
    play_video(frames)

if __name__ == "__main__":
    main()
