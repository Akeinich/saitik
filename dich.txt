using System;
using System.Drawing;
using System.Windows.Forms;

namespace Lab15_Elevator_Designer
{
    public partial class Form1 : Form
    {
        private int floorCount = 5;
        private int currentFloor = 1;
        private bool isMoving = false;
        private int targetFloor = 1;
        private int direction = 0;
        private Timer timer;
        private Button[] floorCallButtons;
        private Button[] cabinButtons;

        public Form1()
        {
            InitializeComponent();
            // Настройка таймера
            timer = new Timer();
            timer.Interval = 300;
            timer.Tick += Timer_Tick;
            // Создаём кнопки для начального количества этажей
            CreateButtonsForFloors(floorCount);
            UpdateDisplay();
        }

        private void CreateButtonsForFloors(int count)
        {
            // Очищаем панели
            panelCallButtons.Controls.Clear();
            panelCabinButtons.Controls.Clear();

            floorCallButtons = new Button[count + 1];
            cabinButtons = new Button[count + 1];

            // Кнопки вызова на этажах (справа)
            for (int i = 1; i <= count; i++)
            {
                Button btn = new Button();
                btn.Text = i.ToString();
                btn.Size = new Size(80, 40);
                btn.Tag = i;
                btn.Click += FloorCallButton_Click;
                floorCallButtons[i] = btn;
                panelCallButtons.Controls.Add(btn);
            }

            // Кнопки в кабине
            for (int i = 1; i <= count; i++)
            {
                Button btn = new Button();
                btn.Text = i.ToString();
                btn.Size = new Size(60, 60);
                btn.Tag = i;
                btn.Click += CabinButton_Click;
                btn.Enabled = false; // пока лифт не активен? (можно true)
                cabinButtons[i] = btn;
                panelCabinButtons.Controls.Add(btn);
            }
        }

        private void btnSetFloors_Click(object sender, EventArgs e)
        {
            // Запрос количества этажей
            string input = Microsoft.VisualBasic.Interaction.InputBox(
                "Введите количество этажей (1-15):", 
                "Этажность", 
                floorCount.ToString());
            if (int.TryParse(input, out int newCount) && newCount >= 1 && newCount <= 15)
            {
                floorCount = newCount;
                CreateButtonsForFloors(floorCount);
                currentFloor = 1;
                targetFloor = 1;
                isMoving = false;
                direction = 0;
                if (timer.Enabled) timer.Stop();
                UpdateDisplay();
            }
            else
            {
                MessageBox.Show("Некорректное число этажей (1-15)");
            }
        }

        private void FloorCallButton_Click(object sender, EventArgs e)
        {
            Button btn = (Button)sender;
            int floor = (int)btn.Tag;
            if (floor == currentFloor && !isMoving)
            {
                MessageBox.Show($"Лифт уже на этаже {floor}");
                return;
            }
            CallElevator(floor);
        }

        private void CabinButton_Click(object sender, EventArgs e)
        {
            Button btn = (Button)sender;
            int floor = (int)btn.Tag;
            if (floor == currentFloor && !isMoving)
            {
                MessageBox.Show($"Вы уже на этаже {floor}");
                return;
            }
            CallElevator(floor);
        }

        private void CallElevator(int floor)
        {
            if (isMoving)
            {
                MessageBox.Show("Лифт уже движется, подождите.");
                return;
            }
            targetFloor = floor;
            if (targetFloor > currentFloor) direction = 1;
            else if (targetFloor < currentFloor) direction = -1;
            else direction = 0;

            if (direction != 0)
            {
                isMoving = true;
                timer.Start();
                SetCabinButtonsEnabled(false);
            }
        }

        private void Timer_Tick(object sender, EventArgs e)
        {
            if (!isMoving) return;

            int newFloor = currentFloor + direction;
            if (direction == 1 && newFloor <= floorCount)
                currentFloor = newFloor;
            else if (direction == -1 && newFloor >= 1)
                currentFloor = newFloor;
            else
            {
                StopElevator();
                return;
            }

            UpdateDisplay();
            if (currentFloor == targetFloor)
                StopElevator();
        }

        private void StopElevator()
        {
            timer.Stop();
            isMoving = false;
            direction = 0;
            UpdateDisplay();
            SetCabinButtonsEnabled(true);
            MessageBox.Show($"Лифт прибыл на этаж {currentFloor}");
        }

        private void SetCabinButtonsEnabled(bool enabled)
        {
            for (int i = 1; i <= floorCount; i++)
                if (cabinButtons[i] != null)
                    cabinButtons[i].Enabled = enabled;
        }

        private void UpdateDisplay()
        {
            lblCurrentFloor.Text = $"Этаж: {currentFloor}";
            panelElevator.BackColor = isMoving ? Color.Yellow : Color.Gray;
        }
    }
}
